# 市场价格方案 - 图片上传示例代码

## 📸 小程序端图片上传完整示例

### **1. 新增设备方案（带图片）**

```javascript
/**
 * 新增设备方案（带图片上传）
 */
addDevicePlanWithImage() {
  const that = this;
  
  // 第一步：选择图片
  wx.chooseImage({
    count: 1,                          // 只能选择1张图片
    sizeType: ['compressed'],          // 压缩图
    sourceType: ['album', 'camera'],   // 可以从相册或拍照
    success(res) {
      const tempFilePath = res.tempFilePaths[0];
      
      // 显示加载提示
      wx.showLoading({
        title: '上传中...',
        mask: true
      });
      
      // 第二步：上传图片
      wx.uploadFile({
        url: 'https://grainservice.club:8443/nongxinle/api/marketpriceplan/admin/addPlanWithImage',
        filePath: tempFilePath,
        name: 'file',                  // 后端接收的参数名
        formData: {
          // 必填参数
          nxMppMarketId: 1,            // 市场ID
          nxMppType: 1,                // 类型：1-设备
          nxMppPlanName: '便携式打印机',
          nxMppPrice: 28000,           // 价格（单位：分）280元=28000分
          
          // 可选参数
          nxMppQuantity: '1台',
          nxMppUnitPrice: '便携式打印机',
          nxMppDescription: '订单打印、标签打印、收据打印',
          nxMppSortOrder: 1,
          nxMppStatus: 1
        },
        success(uploadRes) {
          wx.hideLoading();
          
          const data = JSON.parse(uploadRes.data);
          if (data.code === 0) {
            wx.showToast({
              title: '新增成功',
              icon: 'success'
            });
            
            console.log('新增的方案：', data.data);
            // 刷新列表
            that.loadPlanList();
          } else {
            wx.showToast({
              title: data.msg || '新增失败',
              icon: 'none'
            });
          }
        },
        fail(err) {
          wx.hideLoading();
          wx.showToast({
            title: '上传失败',
            icon: 'none'
          });
          console.error('上传失败：', err);
        }
      });
    }
  });
}
```

### **4. 显示图片（正确的图片路径）**

```javascript
/**
 * 获取完整的图片URL
 * @param {String} imagePath - 数据库中保存的图片路径，如："uploadImage/xxx.png"
 * @returns {String} 完整的图片URL
 */
getImageUrl(imagePath) {
  if (!imagePath) {
    return ''; // 或者返回默认图片
  }
  
  // 拼接完整的图片URL
  const baseUrl = 'https://grainservice.club:8443/nongxinle';
  return `${baseUrl}/${imagePath}`;
}

/**
 * 在WXML中使用
 */
// WXML:
// <image src="{{getImageUrl(item.nxMppImageUrl)}}" mode="aspectFill"></image>

// JS:
data: {
  planList: []
},

onLoad() {
  this.loadPlanList();
},

loadPlanList() {
  wx.request({
    url: 'https://grainservice.club:8443/nongxinle/api/marketpriceplan/miniprogram/getMarketPlans',
    method: 'GET',
    data: {
      marketId: 1,
      type: 1
    },
    success: (res) => {
      if (res.data.code === 0) {
        // 处理图片路径
        const planList = res.data.data.map(item => {
          return {
            ...item,
            imageUrl: this.getImageUrl(item.nxMppImageUrl)
          };
        });
        
        this.setData({
          planList: planList
        });
      }
    }
  });
}
```

#### **JavaScript**
```javascript
// pages/pricePlanEdit/pricePlanEdit.js
Page({
  data: {
    formData: {
      nxMppMarketId: 1,
      nxMppType: 0,
      nxMppPlanName: '',
      nxMppPrice: 0
    },
    priceYuan: '',           // 页面显示的价格（元）
    tempFilePath: '',        // 临时图片路径
    imageUrl: ''             // 图片预览URL
  },

  onLoad(options) {
    // 如果是编辑模式，加载数据
    if (options.id) {
      this.loadPlanDetail(options.id);
    }
  },

  // 类型改变
  onTypeChange(e) {
    this.setData({
      'formData.nxMppType': parseInt(e.detail.value)
    });
  },

  // 名称输入
  onNameInput(e) {
    this.setData({
      'formData.nxMppPlanName': e.detail.value
    });
  },

  // 价格输入（元）
  onPriceInput(e) {
    const yuan = e.detail.value;
    this.setData({
      priceYuan: yuan,
      'formData.nxMppPrice': Math.round(parseFloat(yuan) * 100) // 转为分
    });
  },

  // 选择图片
  chooseImage() {
    wx.chooseImage({
      count: 1,
      sizeType: ['compressed'],
      sourceType: ['album', 'camera'],
      success: (res) => {
        this.setData({
          tempFilePath: res.tempFilePaths[0],
          imageUrl: res.tempFilePaths[0]  // 显示预览
        });
      }
    });
  },

  // 保存方案
  savePlan() {
    const { formData, tempFilePath } = this.data;
    
    // 验证必填项
    if (!formData.nxMppPlanName) {
      wx.showToast({ title: '请输入方案名称', icon: 'none' });
      return;
    }
    if (!formData.nxMppPrice || formData.nxMppPrice <= 0) {
      wx.showToast({ title: '请输入正确的价格', icon: 'none' });
      return;
    }
    
    // 设备类型需要上传图片
    if (formData.nxMppType == 1) {
      if (!tempFilePath) {
        wx.showToast({ title: '请选择设备图片', icon: 'none' });
        return;
      }
      this.uploadWithImage();
    } else {
      // 流量类型不需要图片
      this.uploadWithoutImage();
    }
  },

  // 带图片上传
  uploadWithImage() {
    const { formData, tempFilePath } = this.data;
    
    wx.showLoading({ title: '上传中...', mask: true });
    
    wx.uploadFile({
      url: 'https://grainservice.club:8443/nongxinle/api/marketpriceplan/admin/addPlanWithImage',
      filePath: tempFilePath,
      name: 'file',
      formData: {
        nxMppMarketId: formData.nxMppMarketId,
        nxMppType: formData.nxMppType,
        nxMppPlanName: formData.nxMppPlanName,
        nxMppPrice: formData.nxMppPrice,
        nxMppStatus: 1
      },
      success: (res) => {
        wx.hideLoading();
        const data = JSON.parse(res.data);
        if (data.code === 0) {
          wx.showToast({ title: '保存成功', icon: 'success' });
          setTimeout(() => {
            wx.navigateBack();
          }, 1500);
        } else {
          wx.showToast({ title: data.msg, icon: 'none' });
        }
      },
      fail: () => {
        wx.hideLoading();
        wx.showToast({ title: '上传失败', icon: 'none' });
      }
    });
  },

  // 不带图片上传
  uploadWithoutImage() {
    const { formData } = this.data;
    
    wx.showLoading({ title: '提交中...', mask: true });
    
    wx.request({
      url: 'https://grainservice.club:8443/nongxinle/api/marketpriceplan/admin/addPlan',
      method: 'POST',
      header: { 'Content-Type': 'application/json' },
      data: {
        ...formData,
        nxMppStatus: 1
      },
      success: (res) => {
        wx.hideLoading();
        if (res.data.code === 0) {
          wx.showToast({ title: '保存成功', icon: 'success' });
          setTimeout(() => {
            wx.navigateBack();
          }, 1500);
        } else {
          wx.showToast({ title: res.data.msg, icon: 'none' });
        }
      },
      fail: () => {
        wx.hideLoading();
        wx.showToast({ title: '请求失败', icon: 'none' });
      }
    });
  }
});
```

## 🔑 关键要点

### **1. 图片路径处理**
- **数据库存储**：`uploadImage/xxx.png`（相对路径）
- **前端访问**：`https://grainservice.club:8443/nongxinle/uploadImage/xxx.png`（完整URL）

### **2. 价格单位转换**
- **前端输入**：元（如：280元）
- **后端存储**：分（28000分）
- **转换公式**：`分 = 元 × 100`

### **3. 接口选择**
| 场景 | 接口 | 请求方式 |
|------|------|----------|
| 流量方案（无图片） | `/admin/addPlan` | POST (JSON) |
| 设备方案（有图片） | `/admin/addPlanWithImage` | POST (multipart/form-data) |

### **4. 错误处理**
```javascript
// 统一的错误处理函数
function handleError(res) {
  if (res.data && res.data.code === 0) {
    return true;
  }
  
  wx.showToast({
    title: res.data?.msg || '操作失败',
    icon: 'none',
    duration: 2000
  });
  return false;
}
```

## 📝 注意事项

1. **图片上传时必须使用 `wx.uploadFile`**，不能用 `wx.request`
2. **formData 中的所有值都会被转为字符串**，后端会自动转换类型
3. **图片路径保存时不要加域名**，只保存相对路径
4. **价格一定要转换为分（整数）**，避免精度问题
5. **上传前先验证必填项**，提升用户体验
